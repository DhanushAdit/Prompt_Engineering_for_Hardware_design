            if (SubArith) begin
                ALUResult = (A < B) ? 32'h0000_0001 : 32'h0000_0000;
            end else begin
                ALUResult = 32'h0000_0000;
            end
        end
        3'b100: begin
            ALUResult = A ^ B;
        end
        3'b101: begin
            if (SubArith) begin
                ALUResult = $signed(A) >>> B[4:0];
            end else begin
                ALUResult = A >> B[4:0];
            end
        end
        3'b110: begin
            ALUResult = A | B;
        end
        3'b111: begin
            ALUResult = A & B;
        end
        default: begin
            ALUResult = 32'h0000_0000;
        end
    endcase
end
